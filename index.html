-- NEXORA: bot replies to greetings (hi / hello / namaste) create or replace function public.nx_sup_bot(p_text text) returns text language sql stable security definer set search_path = public as $$ select case when lower(trim(p_text)) ~ '^(hi+|hello+|hey+|hlo|hii|namaste|namaskar|good (morning|afternoon|evening)|salam|assalamualaikum)[\s!.,?]*$' then 'Hello! How can I help you today? You can ask about deposit, withdraw, KYC, P2P, 2FA, trading or fees, or tap "Talk to an agent".' when lower(trim(p_text)) ~ '^(thanks|thank you|thx|ok|okay|ok thanks|dhanyavad)[\s!.,]*$' then 'You are welcome! Ask me anything else, any time.' else (select f.answer from public.nx_faq f where f.enabled and exists (select 1 from unnest(f.keywords) k where lower(p_text) like '%' || lower(k) || '%') order by (select count(*) from unnest(f.keywords) k where lower(p_text) like '%' || lower(k) || '%') desc, (select max(length(k)) from unnest(f.keywords) k where lower(p_text) like '%' || lower(k) || '%') desc, f.sort limit 1) end; $$; revoke all on function public.nx_sup_bot(text) from public, anon, authenticated; select public.nx_sup_bot('Hi') as test;
